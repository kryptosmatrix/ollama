1911: 	// The servers' own words, ahead of the conversation. A tool definition says

1913: 	// for, which is what decides whether a model reaches for it unprompted.

1916: 	}

1918: 		// Skip empty messages if present
1919: 		if m.Content == "" && m.Thinking == "" && len(m.ToolCalls) == 0 && len(m.Attachments) == 0 {
1920: 			continue
1921: 		}
1922: 
1923: 		apiMsg := api.Message{Role: m.Role, Thinking: m.Thinking}
1924: 
1925: 		sb := strings.Builder{}
1926: 		sb.WriteString(m.Content)
1927: 
1928: 		var images []api.ImageData
1929: 		if m.Role == "user" && len(m.Attachments) > 0 {
1930: 			for _, a := range m.Attachments {
1931: 				if isImageAttachment(a.Filename) {
1932: 					images = append(images, api.ImageData(a.Data))
1933: 				} else {
1934: 					content := convertBytesToText(a.Data, a.Filename)
1935: 					sb.WriteString(fmt.Sprintf("\n--- File: %s ---\n%s\n--- End of %s ---",
1936: 						a.Filename, content, a.Filename))
1937: 				}
1938: 			}
1939: 		}
1940: 
1941: 		apiMsg.Content = sb.String()
1942: 		apiMsg.Images = images
1943: 
1944: 		switch m.Role {
1945: 		case "assistant":
1946: 			if len(m.ToolCalls) > 0 {
1947: 				var toolCalls []api.ToolCall
1948: 				for _, tc := range m.ToolCalls {
1949: 					var args api.ToolCallFunctionArguments
1950: 					if err := json.Unmarshal([]byte(tc.Function.Arguments), &args); err != nil {
1951: 						s.log().Error("failed to parse tool call arguments", "error", err, "function_name", tc.Function.Name, "arguments", tc.Function.Arguments)
1952: 						continue
1953: 					}
1954: 
1955: 					toolCalls = append(toolCalls, api.ToolCall{
1956: 						Function: api.ToolCallFunction{
1957: 							Name:      tc.Function.Name,
1958: 							Arguments: args,
1959: 						},
1960: 					})
1961: 				}
1962: 				apiMsg.ToolCalls = toolCalls

1964: 		case "tool":

1967: 			apiMsg.ToolName = m.ToolName

1970: 		default:
1971: 			// Log unknown roles but still include them
1972: 			s.log().Debug("unknown message role", "role", m.Role)
1973: 		}

1976: 	}

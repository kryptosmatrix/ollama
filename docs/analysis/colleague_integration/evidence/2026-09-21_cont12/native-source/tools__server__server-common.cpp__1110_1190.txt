1110:         llama_params["stop"].push_back(stop);
1111:     }
1112:     if (!chat_params.parser.empty()) {
1113:         llama_params["chat_parser"] = chat_params.parser;
1114:     }
1115: 
1116:     llama_params["message_delimiters"] = chat_params.message_delimiters.to_json();
1117: 
1118:     // Reasoning budget: pass parameters through to sampling layer
1119:     {
1120:         int reasoning_budget = json_value(body, "reasoning_budget_tokens",
1121:                                json_value(body, "thinking_budget_tokens", -1));
1122:         if (reasoning_budget == -1) {
1123:             reasoning_budget = opt.reasoning_budget;
1124:         }
1125: 
1126:         if (!chat_params.thinking_end_tag.empty()) {
1127:             llama_params["reasoning_budget_tokens"] = reasoning_budget;
1128:             llama_params["reasoning_budget_start_tag"] = chat_params.thinking_start_tag;
1129:             llama_params["reasoning_budget_end_tag"] = chat_params.thinking_end_tag;
1130:             llama_params["reasoning_budget_message"] = json_value(body, "reasoning_budget_message", opt.reasoning_budget_message);
1131:             llama_params["reasoning_control"] = json_value(body, "reasoning_control", false);
1132:         }
1133:     }
1134: 
1135:     // Handle "logprobs" field
1136:     // TODO: The response format of this option is not yet OAI-compatible, but seems like no one really using it; We may need to fix it in the future
1137:     if (json_value(body, "logprobs", false)) {
1138:         if (has_tools && stream) {
1139:             throw std::invalid_argument("logprobs is not supported with tools + stream");
1140:         }
1141:         llama_params["n_probs"] = json_value(body, "top_logprobs", 20);
1142:     } else if (body.contains("top_logprobs") && !body.at("top_logprobs").is_null()) {
1143:         throw std::invalid_argument("top_logprobs requires logprobs to be set to true");
1144:     }
1145: 
1146:     // Copy remaining properties to llama_params
1147:     // This allows user to use llama.cpp-specific params like "mirostat", ... via OAI endpoint.
1148:     // See "launch_slot_with_task()" for a complete list of params supported by llama.cpp
1149:     for (const auto & item : body.items()) {
1150:         // Exception: if "n_predict" is present, we overwrite the value specified earlier by "max_tokens"
1151:         if (!llama_params.contains(item.key()) || item.key() == "n_predict") {
1152:             llama_params[item.key()] = item.value();
1153:         }
1154:     }
1155: 
1156:     return llama_params;
1157: }

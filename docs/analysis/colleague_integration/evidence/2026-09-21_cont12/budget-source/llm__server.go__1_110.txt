36: func filteredEnvLogKey(key string) bool {
37: 	return strings.HasPrefix(key, "CUDA_") ||
38: 		strings.HasPrefix(key, "ROCR_") ||
39: 		strings.HasPrefix(key, "ROCM_") ||
40: 		strings.HasPrefix(key, "HIP_") ||
41: 		strings.HasPrefix(key, "HSA_") ||
42: 		strings.HasPrefix(key, "GGML_") ||
43: 		slices.Contains([]string{
44: 			"PATH",
45: 			"LD_LIBRARY_PATH",
46: 			"DYLD_LIBRARY_PATH",

87: // LoadModel will load a model from disk. The model must be in the GGML format.
88: //
89: // It collects array values for arrays with a size less than or equal to
90: // maxArraySize. If maxArraySize is 0, the default value of 1024 is used. If
91: // the maxArraySize is negative, all arrays are collected.
92: func LoadModel(model string, maxArraySize int) (*ggml.GGML, error) {
93: 	if _, err := os.Stat(model); err != nil {
94: 		return nil, err
95: 	}
96: 
97: 	f, err := os.Open(model)
98: 	if err != nil {
99: 		return nil, err
100: 	}
101: 	defer f.Close()
102: 
103: 	return ggml.Decode(f, maxArraySize)
104: }
